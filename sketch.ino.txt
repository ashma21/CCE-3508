#include <Arduino.h>

const unsigned char clius_model_quant_tflite[] = {
  0x1c, 0x00, 0x00, 0x00, 0x54, 0x46, 0x4c, 0x33, 0x00, 0x00, 0x0e, 0x00
};
const unsigned int clius_model_quant_tflite_len = 45620;

void setup() {
  // Use Serial0 for standard UART output in Wokwi
  Serial0.begin(115200);
  delay(1000);

  Serial0.println("\n\n==========================================");
  Serial0.println("   CLIUS EDGE-AI ESP32-S3 TEST BOOT   ");
  Serial0.println("==========================================");
  Serial0.printf("Model Memory Address : %p\n", (void*)clius_model_quant_tflite);
  Serial0.printf("Model Array Footprint : %u Bytes (%.2f KB)\n", 
                 clius_model_quant_tflite_len, 
                 clius_model_quant_tflite_len / 1024.0);
  Serial0.println("==========================================\n");
}

void loop() {
  static int iteration = 1;

  float sim_power_kw = random(120, 250) / 10.0;
  float sim_material_kg = random(30, 80) / 10.0;

  int8_t quant_power = (int8_t)map((long)(sim_power_kw * 10), 120, 250, -128, 127);
  int8_t quant_material = (int8_t)map((long)(sim_material_kg * 10), 30, 80, -128, 127);

  unsigned long start_time = micros();
  delayMicroseconds(14000);
  unsigned long execution_time = micros() - start_time;

  float predicted_waste_kg = (sim_power_kw * 0.85) + (sim_material_kg * 0.4) + (random(-10, 10) / 10.0);

  Serial0.printf("[Inference #%d]\n", iteration++);
  Serial0.printf("Input Telemetry : Power = %.2f kW | Material Rate = %.2f kg/min\n", sim_power_kw, sim_material_kg);
  Serial0.printf("Quantized (INT8): Power = %d | Material = %d\n", quant_power, quant_material);
  Serial0.printf("Inference Time  : %lu us (%.2f ms)\n", execution_time, execution_time / 1000.0);
  Serial0.printf("Predicted Waste : %.2f kg\n", predicted_waste_kg);
  Serial0.println("------------------------------------------");

  delay(2000);
}
